    53→// PIN Screen
    54→// ========================
    55→function PinScreen({ onAccess }: { onAccess: () => void }) {
    56→  const [pin, setPin] = useState('')
    57→  const [shake, setShake] = useState(false)
    58→  const CORRECT_PIN = '233023'
    59→
    60→  const handlePinInput = (value: string) => {
    61→    if (pin.length < 6) {
    62→      const newPin = pin + value
    63→      setPin(newPin)
    64→      if (newPin.length === 6) {
    65→        if (newPin === CORRECT_PIN) {
    66→          setTimeout(() => onAccess(), 200)
    67→        } else {
    68→          setShake(true)
    69→          setTimeout(() => {
    70→            setPin('')
    71→            setShake(false)
    72→          }, 500)
    73→        }
    74→      }
    75→    }
    76→  }
    77→
    78→  const handleDelete = () => {
    79→    setPin(prev => prev.slice(0, -1))
    80→  }
    81→
    82→  return (
    83→    <div className="min-h-screen overflow-x-hidden bg-gradient-to-b from-zinc-950 via-zinc-900 to-zinc-950 flex flex-col items-center justify-center px-4 pb-safe">
    84→      <div className="mb-10 text-center">
    85→        <div className="relative inline-block">
    86→          <div className="absolute inset-0 bg-emerald-500/20 rounded-full blur-xl scale-150" />
    87→          <div className="relative bg-zinc-800/80 rounded-2xl p-4 border border-zinc-700/50">
    88→            <Package className="w-10 h-10 text-emerald-400 mx-auto" />
    89→          </div>
    90→        </div>
    91→        <h1 className="text-white text-xl font-bold mt-5 tracking-tight">Estoque</h1>
    92→        <p className="text-zinc-500 text-sm mt-1.5">Digite o PIN para acessar</p>
    93→      </div>
    94→
    95→      <div className={`flex gap-4 mb-10 ${shake ? 'animate-[shake_0.5s_ease-in-out]' : ''}`}>
    96→        {Array.from({ length: 6 }).map((_, i) => (
    97→          <div
    98→            key={i}
    99→            className={`w-4 h-4 rounded-full transition-all duration-300 ${
   100→              i < pin.length
   101→                ? 'bg-emerald-400 shadow-[0_0_12px_rgba(52,211,153,0.6)] scale-125'
   102→                : 'bg-zinc-700'
   103→            }`}
   104→          />
   105→        ))}
   106→      </div>
   107→
   108→      <div className="grid grid-cols-3 gap-3 w-full max-w-[300px]">
   109→        {['1', '2', '3', '4', '5', '6', '7', '8', '9', '', '0', 'del'].map((key) => (
   110→          <button
   111→            key={key || 'empty'}
   112→            className={`h-[4.5rem] text-xl font-medium rounded-2xl transition-all active:scale-95 ${
   113→              key === 'del'
   114→                ? 'text-red-400 hover:text-red-300 hover:bg-zinc-800/80 active:bg-zinc-700'
   115→                : key
   116→                  ? 'text-white hover:bg-zinc-800/80 bg-zinc-800/50 border border-zinc-700/50 active:bg-zinc-700'
   117→                  : 'invisible'
   118→            }`}
   119→            onClick={() => {
   120→              if (key === 'del') handleDelete()
   121→              else if (key) handlePinInput(key)
   122→            }}
   123→          >
   124→            {key === 'del' ? <X className="w-6 h-6 mx-auto" /> : key}
   125→          </button>
   126→        ))}
   127→      </div>
   128→
   129→      <style>{`
   130→        @keyframes shake {
   131→          0%, 100% { transform: translateX(0); }
   132→          20% { transform: translateX(-10px); }
   133→          40% { transform: translateX(10px); }
   134→          60% { transform: translateX(-10px); }
   135→          80% { transform: translateX(10px); }
   136→        }
   137→      `}</style>
   138→    </div>
   139→  )
   140→}

   402→function ProductsScreen({ onBack }: { onBack: () => void }) {
   403→  const [products, setProducts] = useState<Product[]>([])
   404→  const [name, setName] = useState('')
   405→  const [stock, setStock] = useState('')
   406→  const [editingId, setEditingId] = useState<string | null>(null)
   407→  const [loading, setLoading] = useState(false)
   408→  const { toast } = useToast()
   409→
   410→  const [, startTransition] = useTransition()
   411→
   412→  const loadProducts = useCallback(async () => {
   413→    try {
   414→      const res = await fetch('/api/products')
   415→      const data = await res.json()
   416→      if (Array.isArray(data)) {
   417→        startTransition(() => setProducts(data))
   418→      }
   419→    } catch {
   420→      toast({ title: 'Erro ao carregar produtos', variant: 'destructive' })
   421→    }
   422→  }, [toast, startTransition])
   423→
   424→  useEffect(() => { loadProducts() }, [loadProducts])
   425→
   426→  const handleSubmit = async () => {
   427→    if (!name.trim()) {
   428→      toast({ title: 'Nome é obrigatório', variant: 'destructive' })
   429→      return
   430→    }
   431→
   432→    setLoading(true)
   433→    try {
   434→      if (editingId) {
   435→        const res = await fetch('/api/products', {
   436→          method: 'PUT',
   437→          headers: { 'Content-Type': 'application/json' },
   438→          body: JSON.stringify({ id: editingId, name, stock: stock || undefined }),
   439→        })
   440→        if (!res.ok) {
   441→          const data = await res.json()
   442→          toast({ title: data.error || 'Erro ao atualizar', variant: 'destructive' })
   443→          setLoading(false)
   444→          return
   445→        }
   446→        toast({ title: 'Produto atualizado!' })
   447→        setEditingId(null)
   448→      } else {
   449→        const res = await fetch('/api/products', {
   450→          method: 'POST',
   451→          headers: { 'Content-Type': 'application/json' },
   452→          body: JSON.stringify({ name }),
   453→        })
   454→        if (!res.ok) {
   455→          const data = await res.json()
   456→          toast({ title: data.error || 'Erro ao cadastrar', variant: 'destructive' })
   457→          setLoading(false)
   458→          return
   459→        }
   460→        toast({ title: 'Produto cadastrado!' })
   461→      }
   462→      setName('')
   463→      setStock('')
   464→      loadProducts()
   465→    } catch {
   466→      toast({ title: 'Erro ao salvar', variant: 'destructive' })
   467→    }
   468→    setLoading(false)
   469→  }
   470→
   471→  const handleEdit = (p: Product) => {
   472→    setEditingId(p.id)
   473→    setName(p.name)
   474→    setStock(p.stock > 0 ? p.stock.toString() : '')
   475→  }
   476→
   477→  const handleDelete = async (id: string) => {
   478→    if (!confirm('Deletar este produto e todas suas entradas/movimentações?')) return
   479→    try {
   480→      const res = await fetch(`/api/products?id=${id}`, { method: 'DELETE' })
   481→      if (!res.ok) {
   482→        const data = await res.json()
   483→        toast({ title: data.error || 'Erro ao deletar', variant: 'destructive' })
   484→        return
   485→      }
   486→      toast({ title: 'Produto removido' })
   487→      loadProducts()
   488→    } catch {
   489→      toast({ title: 'Erro ao deletar', variant: 'destructive' })
   490→    }
   491→  }
   492→
   493→  const cancelEdit = () => {
   494→    setEditingId(null)
   495→    setName('')
   496→    setStock('')
   497→  }
   498→
   499→  return (
   500→    <div className="min-h-screen overflow-x-hidden bg-zinc-50 pb-8">
   501→      <header className="bg-gradient-to-br from-zinc-950 via-zinc-900 to-zinc-800 text-white px-4 py-4 pt-safe">
   502→        <div className="flex items-center gap-3">
   503→          <Button variant="ghost" size="icon" className="text-white hover:bg-white/10 -ml-2" onClick={onBack}>

   610→function MovementScreen({

   612→  onBack,
   613→  onComplete,

   615→  type: 'ENTRADA' | 'SAIDA'
   616→  onBack: () => void

   619→  const isEntrada = type === 'ENTRADA'
   620→  const [products, setProducts] = useState<Product[]>([])
   621→  const [quantities, setQuantities] = useState<Record<string, string>>({})
   622→  const [costPrices, setCostPrices] = useState<Record<string, string>>({})
   623→  const [salePrices, setSalePrices] = useState<Record<string, string>>({})
   624→  const [averageCosts, setAverageCosts] = useState<Record<string, number>>({})
   625→  const [clientName, setClientName] = useState('')
   626→  const [search, setSearch] = useState('')
   627→  const [loading, setLoading] = useState(false)
   628→  const { toast } = useToast()
   629→
   630→  useEffect(() => {
   631→    fetch('/api/products').then((r) => r.json()).then((data: Product[]) => {
   632→      if (!Array.isArray(data)) return
   633→      setProducts(data)
   634→      const avgs: Record<string, number> = {}
   635→      data.forEach((p) => { avgs[p.id] = p.averageCost })
   636→      setAverageCosts(avgs)
   637→    })
   638→  }, [])
   639→
   640→  const filtered = products.filter((p) =>
   641→    p.name.toLowerCase().includes(search.toLowerCase())

   694→      if (missingCost) {
   695→        const p = products.find(pr => pr.id === missingCost.productId)
   696→        toast({ title: `Informe o custo para ${p?.name || 'o produto'}`, variant: 'destructive' })
   697→        return
   698→      }
   699→    }
   700→
   701→    if (!isEntrada) {
   702→      const missingSale = selectedItems.find(i => !i.salePrice || parseFloat(i.salePrice) <= 0)
   703→      if (missingSale) {
   704→        const p = products.find(pr => pr.id === missingSale.productId)
   705→        toast({ title: `Informe o preço de venda para ${p?.name || 'o produto'}`, variant: 'destructive' })
   706→        return
   707→      }
   708→    }
   709→
   710→    setLoading(true)
   711→    try {
   712→      const res = await fetch('/api/movements', {
   713→        method: 'POST',
   714→        headers: { 'Content-Type': 'application/json' },
   715→        body: JSON.stringify({
   716→          type,
   717→          items: selectedItems,
   718→          clientName: clientName.trim(),
   719→        }),
   720→      })
   721→      const data = await res.json()
   722→
   723→      if (!res.ok) {

   725→        setLoading(false)
   726→        return
   727→      }

   733→    }

   735→  }

   738→    ? { active: 'bg-emerald-50 border-emerald-300', btn: 'bg-emerald-500 hover:bg-emerald-600 text-white', badge: 'bg-emerald-500/15 text-emerald-700' }

   744→        isEntrada
   745→          ? 'bg-gradient-to-br from-emerald-600 to-emerald-700'

   749→          <div className="flex items-center gap-3 min-w-0">
   750→            <Button variant="ghost" size="icon" className="text-white hover:bg-white/15 -ml-2 shrink-0" onClick={onBack}>
   751→              <X className="w-5 h-5" />
   752→            </Button>
   753→            <div className="min-w-0">
   754→              <h1 className="text-lg font-bold truncate">
   755→                {isEntrada ? 'Entrada' : 'Venda'}
   756→              </h1>
   757→              <p className="text-[11px] opacity-70">
   758→                {isEntrada ? 'Registrar entrada' : 'Registrar venda'}
   759→              </p>
   760→            </div>
   761→          </div>
   762→          <div className="text-right shrink-0">
   763→            <p className="text-xl font-bold tabular-nums truncate">

   766→            <p className="text-[11px] opacity-70">
   767→              {totalItems} {totalItems === 1 ? 'item' : 'itens'}
   768→            </p>
   769→          </div>
   770→        </div>
   771→      </header>
   772→
   773→      <div className="w-full px-4 pt-4 space-y-3 flex-1 flex flex-col min-h-0">
   774→        <div className="flex items-center gap-2.5 bg-white border border-zinc-200 shadow-sm rounded-xl px-3.5 py-3 shrink-0">
   775→          <User className={`w-4 h-4 shrink-0 ${isEntrada ? 'text-emerald-500' : 'text-red-500'}`} />
   776→          <input
   777→            type="text"
   778→            placeholder={isEntrada ? 'Nome do fornecedor' : 'Nome do cliente'}
   779→            value={clientName}
   780→            onChange={(e) => setClientName(e.target.value)}
   781→            className="flex-1 outline-none text-sm bg-transparent min-w-0"
   782→          />
   783→        </div>
   784→
   785→        <div className="shrink-0">
   786→          <Input
   787→            placeholder="Buscar produto..."
   788→            value={search}
   789→            onChange={(e) => setSearch(e.target.value)}

   799→              </p>

   801→            {filtered.map((p) => {

   804→              const avgCost = averageCosts[p.id] || 0

   896→                        onClick={() => adjustQty(p.id, 1)}
   897→                        className={`w-10 h-10 rounded-xl flex items-center justify-center transition-colors ${accentColor.btn}`}
   898→                      >
   899→                        <Plus className="w-4 h-4" />
   900→                      </button>
   901→                    </div>
   902→                  </div>
   903→                </div>
   904→              )
   905→            })}
   906→          </div>
   907→        </div>
   908→
   909→        {selectedItems.length > 0 && (
   910→          <div className="bg-white border border-zinc-200 shadow-lg rounded-2xl p-4 space-y-3 shrink-0 mt-2">
   911→            <div className="flex justify-between text-sm">
   912→              <span className="text-zinc-500">{selectedItems.length} produto{selectedItems.length !== 1 ? 's' : ''} · {totalItems} {totalItems === 1 ? 'item' : 'itens'}</span>
   913→              <span className="font-bold text-lg tabular-nums">
   914→                R$ {totalValue.toLocaleString('pt-BR', { minimumFractionDigits: 2 })}
   915→              </span>
   916→            </div>
   917→            <Button
   918→              onClick={handleSubmit}
   919→              disabled={loading}
   920→              className={`w-full h-12 text-base font-semibold text-white rounded-xl ${
   921→                isEntrada
   922→                  ? 'bg-emerald-600 hover:bg-emerald-700'
   923→                  : 'bg-red-500 hover:bg-red-600'
   924→              }`}
   925→            >
   926→              {loading
   927→                ? 'Registrando...'
   928→                : isEntrada
   929→                  ? 'Confirmar Entrada'
   930→                  : 'Confirmar Venda'}
   931→            </Button>
   932→          </div>
   933→        )}
   934→      </div>
   935→    </div>
   936→  )
   937→}

   942→function CupomScreen({
   943→  cupomId,
   944→  onBack,
   945→}: {
   946→  cupomId: string
   947→  onBack: () => void
   948→}) {
   949→  const [movements, setMovements] = useState<MovementItem[]>([])
   950→  const [deleted, setDeleted] = useState(false)
   951→  const [deleting, setDeleting] = useState(false)
   952→  const { toast } = useToast()
   953→
   954→  useEffect(() => {
   955→    fetch(`/api/movements?cupomId=${cupomId}`)
   956→      .then((r) => r.json())
   957→      .then((data) => {
   958→        if (Array.isArray(data)) setMovements(data)
   959→      })
   960→  }, [cupomId])
   961→
   962→  const totalGeral = movements.reduce((sum, m) => sum + m.total, 0)
   963→  const totalQtd = movements.reduce((sum, m) => sum + m.quantity, 0)
   964→  const isEntrada = movements[0]?.type === 'ENTRADA'
   965→  const clientName = movements[0]?.clientName
   966→  const date = movements[0]?.createdAt
   967→    ? new Date(movements[0].createdAt).toLocaleString('pt-BR')
   968→    : ''
   969→  const shortId = cupomId.slice(0, 8).toUpperCase()
   970→
   971→  const totalCost = movements.reduce((sum, m) => sum + ((m.averageCost || 0) * m.quantity), 0)
   972→  const profit = totalGeral - totalCost
   973→
   974→  const handleDelete = async () => {
   975→    if (!confirm(`Excluir esta ${movements[0]?.type === 'ENTRADA' ? 'entrada' : 'venda'}?\nO estoque será revertido automaticamente.`)) return
   976→
   977→    setDeleting(true)
   978→    try {
   979→      const res = await fetch(`/api/movements?cupomId=${cupomId}`, { method: 'DELETE' })
   980→      const data = await res.json()
   981→
   982→      if (!res.ok) {
   983→        toast({ title: data.error || 'Erro ao excluir', variant: 'destructive' })
   984→        setDeleting(false)
   985→        return
   986→      }
   987→
   988→      toast({ title: `${movements[0]?.type === 'ENTRADA' ? 'Entrada' : 'Venda'} excluída! Estoque revertido.` })
   989→      setDeleted(true)
   990→    } catch {
   991→      toast({ title: 'Erro ao excluir', variant: 'destructive' })
   992→    }
   993→    setDeleting(false)
   994→  }
   995→
